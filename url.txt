/* ============================================================
   Dynamics 365 解决方案组件导出脚本 v3
   设计理念：不关心UI(classic/modern)怎么展示，只关心 solutioncomponents
   表里"真实"存在的数据。

   相比v2的改动：
   - 类型名称不再靠手写映射表猜测，而是实时查询该环境的
     componenttype 全局选项集(GlobalOptionSetDefinitions)，
     保证类型名称100%准确、和实际环境版本一致
   - 去掉"是否UI可见"这个维度，改为更有意义的"反查状态"：
       成功 / 失败(疑似孤儿引用，指向对象已被删除) / 暂不支持反查
   - 新增部分较新组件类型的反查支持：Custom API、
     Connection Reference、Environment Variable 等
   - 新增 Attribute(字段) 反查：利用 rootsolutioncomponentid 找到
     所属Entity组件，间接定位表名，再反查出"表名.字段名"
     （仅覆盖直接挂在Entity下的字段；其余metadata子组件类型
     如AttributePicklistValue、ViewAttribute等暂不支持，仍标"暂不支持反查"）
   - 只导出一份"真实清单"CSV

   使用方法：CRM/PowerApps 应用页面 -> F12控制台 -> 改solution名 -> 粘贴运行
   ============================================================ */

(async function () {
  const SOLUTION_UNIQUE_NAME = "AIAXRPSLN_2607_1"; // <-- 改成你的solution唯一名

  const apiBase = window.location.origin + "/api/data/v9.2";

  // 已知可反查名称的组件类型：componenttype编号 -> {实体集, 名称字段}
  // 没列在这里的类型，仍然会从componenttype选项集拿到正确的"类型名称"，
  // 只是"名称"这一列会标注"暂不支持反查"，而不是报错或误判。
  const RESOLVE_MAP = {
    1: { set: "EntityDefinitions", nameField: "LogicalName" },
    9: { set: "GlobalOptionSetDefinitions", nameField: "Name" },
    10: { set: "RelationshipDefinitions", nameField: "SchemaName" },
    20: { set: "roles", nameField: "name" },
    26: { set: "savedqueries", nameField: "name", parentField: "returnedtypecode" },
    29: { set: "workflows", nameField: "name" },
    31: { set: "reports", nameField: "name" },
    36: { set: "templates", nameField: "title" },
    44: { set: "duplicaterules", nameField: "name" },
    59: { set: "savedqueryvisualizations", nameField: "name", parentField: "primaryentitytypecode" },
    60: { set: "systemforms", nameField: "name", parentField: "objecttypecode" },
    61: { set: "webresourceset", nameField: "name" },
    62: { set: "sitemaps", nameField: "sitemapnameunique" },
    63: { set: "connectionroles", nameField: "name" },
    65: { set: "hierarchyrules", nameField: "name" },
    70: { set: "fieldsecurityprofiles", nameField: "name" },
    80: { set: "appmodules", nameField: "name" },
    90: { set: "plugintypes", nameField: "name" },
    91: { set: "pluginassemblies", nameField: "name" },
    92: { set: "sdkmessageprocessingsteps", nameField: "name" },
    93: { set: "sdkmessageprocessingstepimages", nameField: "name" },
    95: { set: "serviceendpoints", nameField: "name" },
    136: { set: "customapis", nameField: "uniquename" },
    137: { set: "customapirequestparameters", nameField: "uniquename" },
    138: { set: "customapiresponseproperties", nameField: "uniquename" },
    152: { set: "slas", nameField: "name" },
    161: { set: "mobileofflineprofiles", nameField: "name" },
    162: { set: "mobileofflineprofileitems", nameField: "name" },
    372: { set: "connectionreferences", nameField: "connectionreferencedisplayname" },
    380: { set: "environmentvariabledefinitions", nameField: "displayname" },
  };

  async function getJson(url, retry = true) {
    const res = await fetch(url, {
      headers: { "OData-MaxVersion": "4.0", "OData-Version": "4.0", Accept: "application/json" },
    });
    if (res.status === 429 && retry) {
      await new Promise((r) => setTimeout(r, 1500));
      return getJson(url, false);
    }
    if (!res.ok) throw new Error(`${res.status} ${url}`);
    return res.json();
  }

  console.log("正在获取 componenttype 选项集（真实类型名称来源）...");
  // componenttype 是 solutioncomponent 实体上的本地(字段级)选项集，不是全局选项集，
  // 所以要通过 EntityDefinitions -> Attributes -> PicklistAttributeMetadata 去查，
  // 而不是 GlobalOptionSetDefinitions（那样会400）。
  const componentTypeAttr = await getJson(
    `${apiBase}/EntityDefinitions(LogicalName='solutioncomponent')/Attributes(LogicalName='componenttype')/Microsoft.Dynamics.CRM.PicklistAttributeMetadata?$select=LogicalName&$expand=OptionSet($select=Options)`
  );
  const TYPE_NAME_MAP = {};
  const rawOptions = (componentTypeAttr.OptionSet && componentTypeAttr.OptionSet.Options) || [];
  rawOptions.forEach((opt) => {
    const label =
      (opt.Label && opt.Label.UserLocalizedLabel && opt.Label.UserLocalizedLabel.Label) ||
      `(无标签,值${opt.Value})`;
    TYPE_NAME_MAP[opt.Value] = label;
  });
  console.log(`已获取 ${Object.keys(TYPE_NAME_MAP).length} 个类型定义`);

  function typeName(t) {
    return TYPE_NAME_MAP[t] || `未在选项集中定义(${t})`;
  }

  console.log("正在解析 solution id ...");
  const solResp = await getJson(
    `${apiBase}/solutions?$filter=uniquename eq '${SOLUTION_UNIQUE_NAME}'&$select=solutionid,friendlyname`
  );
  if (!solResp.value.length) {
    console.error("找不到该solution，请检查 SOLUTION_UNIQUE_NAME 是否正确");
    return;
  }
  const solutionId = solResp.value[0].solutionid;
  console.log(`找到 solution: ${solResp.value[0].friendlyname} (${solutionId})`);

  let url = `${apiBase}/solutioncomponents?$filter=_solutionid_value eq ${solutionId}&$select=objectid,componenttype,rootcomponentbehavior,ismetadata,solutioncomponentid,rootsolutioncomponentid`;
  let all = [];
  while (url) {
    const page = await getJson(url);
    all = all.concat(page.value);
    url = page["@odata.nextLink"] || null;
  }
  console.log(`共拉取到 ${all.length} 条 solutioncomponents 原始记录（这就是该solution真实包含的全部内容）`);

  const groups = {};
  for (const c of all) {
    (groups[c.componenttype] = groups[c.componenttype] || []).push(c);
  }

  async function resolveNames(list, mapEntry) {
    if (!mapEntry) {
      return list.map((c) => ({ ...c, resolvedName: "(暂不支持反查)", resolveStatus: "暂不支持反查" }));
    }
    const selectFields = mapEntry.parentField
      ? `${mapEntry.nameField},${mapEntry.parentField}`
      : mapEntry.nameField;
    const chunkSize = 5;
    const results = [];
    for (let i = 0; i < list.length; i += chunkSize) {
      const chunk = list.slice(i, i + chunkSize);
      const promises = chunk.map(async (c) => {
        try {
          const d = await getJson(`${apiBase}/${mapEntry.set}(${c.objectid})?$select=${selectFields}`);
          const baseName = d[mapEntry.nameField] || "(空值)";
          const prefixedName = mapEntry.parentField && d[mapEntry.parentField]
            ? `${d[mapEntry.parentField]}.${baseName}`
            : baseName;
          return { ...c, resolvedName: prefixedName, resolveStatus: "成功" };
        } catch (e) {
          return {
            ...c,
            resolvedName: `(反查失败: ${e.message})`,
            resolveStatus: "失败(疑似孤儿引用/对象已删除)",
          };
        }
      });
      results.push(...(await Promise.all(promises)));
      await new Promise((r) => setTimeout(r, 150));
    }
    return results;
  }

  // componenttype=2(Attribute)这类"挂靠在表下面"的metadata子组件，无法直接用一个
  // 固定实体集反查——必须先知道它属于哪张表。solutioncomponents记录本身不直接带表名，
  // 但每条子组件的 rootsolutioncomponentid 指向的正是它所属Entity组件的 solutioncomponentid，
  // 所以思路是：先解析出所有Entity，建立"solutioncomponentid -> 表LogicalName"的映射，
  // 再用这个映射把Attribute的rootsolutioncomponentid转成表名，最后用表名+objectid去查字段名。
  const ATTRIBUTE_TYPE = 2;

  async function resolveAttributeNames(list, solutionComponentIdToEntityName) {
    const chunkSize = 5;
    const results = [];
    for (let i = 0; i < list.length; i += chunkSize) {
      const chunk = list.slice(i, i + chunkSize);
      const promises = chunk.map(async (c) => {
        const entityLogicalName = solutionComponentIdToEntityName[c.rootsolutioncomponentid];
        if (!entityLogicalName) {
          return {
            ...c,
            resolvedName: "(无法定位所属表)",
            resolveStatus: "失败(找不到父级Entity组件，无法定位所属表)",
          };
        }
        try {
          const d = await getJson(
            `${apiBase}/EntityDefinitions(LogicalName='${entityLogicalName}')/Attributes(${c.objectid})?$select=LogicalName`
          );
          return {
            ...c,
            resolvedName: `${entityLogicalName}.${d.LogicalName}`,
            resolveStatus: "成功",
          };
        } catch (e) {
          return {
            ...c,
            resolvedName: `(反查失败: ${e.message})`,
            resolveStatus: "失败(疑似孤儿引用/字段已删除)",
          };
        }
      });
      results.push(...(await Promise.all(promises)));
      await new Promise((r) => setTimeout(r, 150));
    }
    return results;
  }

  const finalRows = [];
  const solutionComponentIdToEntityName = {};

  // 先处理 Entity(1)，建立 solutioncomponentid -> 表LogicalName 映射，供Attribute反查使用
  const entityType = 1;
  if (groups[entityType]) {
    const resolvedEntities = await resolveNames(groups[entityType], RESOLVE_MAP[entityType]);
    resolvedEntities.forEach((r) => {
      if (r.resolveStatus === "成功") {
        solutionComponentIdToEntityName[r.solutioncomponentid] = r.resolvedName;
      }
      finalRows.push({
        类型编号: String(entityType),
        类型名称: typeName(entityType),
        名称: r.resolvedName,
        反查状态: r.resolveStatus,
        IsMetadata: r.ismetadata,
        ObjectId: r.objectid,
        RootComponentBehavior: r.rootcomponentbehavior,
      });
    });
  }

  for (const [type, list] of Object.entries(groups)) {
    const typeNum = Number(type);
    if (typeNum === entityType) continue; // 已处理

    let resolved;
    if (typeNum === ATTRIBUTE_TYPE) {
      resolved = await resolveAttributeNames(list, solutionComponentIdToEntityName);
    } else {
      resolved = await resolveNames(list, RESOLVE_MAP[typeNum]);
    }

    resolved.forEach((r) =>
      finalRows.push({
        类型编号: type,
        类型名称: typeName(typeNum),
        名称: r.resolvedName,
        反查状态: r.resolveStatus,
        IsMetadata: r.ismetadata,
        ObjectId: r.objectid,
        RootComponentBehavior: r.rootcomponentbehavior,
      })
    );
  }

  console.log("========== 按类型统计（真实全部数量，不受任何UI展示逻辑影响） ==========");
  const summary = {};
  finalRows.forEach((r) => (summary[r.类型名称] = (summary[r.类型名称] || 0) + 1));
  console.table(summary);
  console.log(`solutioncomponents 真实记录总数: ${finalRows.length}`);

  const failedRows = finalRows.filter((r) => r.反查状态.startsWith("失败"));
  if (failedRows.length) {
    console.warn(
      `发现 ${failedRows.length} 条反查失败的记录（疑似孤儿引用），已在CSV中标出，建议核实后考虑清理：`
    );
    console.table(failedRows);
  }

  console.log("========== 完整明细 ==========");
  console.table(finalRows);

  function downloadCsv(rows, filename) {
    const headers = ["类型编号", "类型名称", "名称", "反查状态", "IsMetadata", "ObjectId", "RootComponentBehavior"];
    const csvRows = [headers.join(",")];
    rows.forEach((r) => {
      csvRows.push(headers.map((h) => `"${String(r[h] ?? "").replace(/"/g, '""')}"`).join(","));
    });
    const csvContent = "\uFEFF" + csvRows.join("\n");
    const blob = new Blob([csvContent], { type: "text/csv;charset=utf-8;" });
    const link = document.createElement("a");
    link.href = URL.createObjectURL(blob);
    link.download = filename;
    link.click();
  }

  downloadCsv(finalRows, `${SOLUTION_UNIQUE_NAME}_components_真实清单.csv`);

  console.log("完成！已下载真实清单CSV，包含该solution下solutioncomponents表的全部记录。");
})();
