# 1. 生成 RSA2048 私钥+证书
openssl req \
-newkey rsa:2048 \
-nodes \
-keyout private.key \
-x509 \
-days 365 \
-out certificate.crt \
-subj "/emailAddress=admin@contoso.com/CN=Contoso"

# 2. 导出 PFX（可选）
openssl pkcs12 \
-export \
-out certificate.pfx \
-inkey private.key \
-in certificate.crt

# 3. 导出 CER（DER）
openssl x509 \
-in certificate.crt \
-outform DER \
-out certificate.cer

# 4. 计算 SHA256
openssl dgst -sha256 certificate.cer
